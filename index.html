SQL Query Examples

Example 1: Find Schema–Table–Column value


SELECT 
    TABLE_SCHEMA + '.' + TABLE_NAME + '.' + COLUMN_NAME AS SchemaTableColumn
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA LIKE '%Schema%'
   OR TABLE_NAME LIKE '%Table%'
   OR COLUMN_NAME LIKE '%Column%'
ORDER BY TABLE_SCHEMA, TABLE_NAME, COLUMN_NAME;

Example 2: Verify legacy DOB values converted to new format (yyyy-MM-dd)


SELECT 
   l.Customer_ID,
   l.DOB AS LegacyDOB,
   n.BirthDate AS NewBirthDate
FROM LegacySystem.Customers l
JOIN NewPlatform.Customers n
   ON l.Customer_ID = n.CustomerNumber
WHERE TRY_CONVERT(date, n.BirthDate) IS NULL
   OR FORMAT(l.DOB, 'yyyy-MM-dd') <> n.BirthDate;

Example 3: Find legacy brokers missing from new platform

  
SELECT l.Broker_ID
FROM LegacySystem.Brokers l
LEFT JOIN NewPlatform.Brokers n
   ON l.Broker_ID = n.AgentCode
WHERE n.AgentCode IS NULL;
